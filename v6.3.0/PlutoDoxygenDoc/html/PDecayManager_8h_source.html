43// template PNextList:
44// implements a linked list of pointers(!) to object T. A delete does
45// not imply a delete of the object T itself. If this is wanted -> use
46// member function Delete().
47template<class T>
48class PNextList {
49 public:
50 T *Curr; // Pointer to object T
51 PNextList *Next; // Pointer to next list node
52
53 PNextList() : Curr(NULL), Next(NULL) {};
54 PNextList(T *t) : Curr(t), Next(NULL) {};
55 ~PNextList() { if (Next) delete Next; };
56
57 void Add(T *t) {
58 if (!Curr)
59 Curr = t; // if node still empty
60 else if (!Next)
61 Next = new PNextList<T>(t); // if node has no successor
62 else
63 Next->Add(t);
64 }
65
66 void Delete() { // delete all T objects
67 delete Curr;
68 if (Next)
69 Next->Delete();
70 }
71};
72// end of template PNextList

74// template PStack:
75// implements a last-in-first-out standard stack via a single linked
76// list. Additionally the number of objects is counted.
77template<class T>
78class PStack {
79 public:
80 PNextList<T> *top; // Pointer to top node
81 Int_t Count;
82
83 PStack() : top(NULL), Count(0) {};
84 ~PStack() { delete top; }; // delete the nodes, not
85 // the object itself
86 void Push(T* t) {
87 PNextList<T> *temp = new PNextList<T>(t); // new node with object T
88 temp->Next = top; // position before top node
89 top = temp; // make new node the top node
90 Count++;
91 }
92
93 T *Pop() {
94 if (!top) return NULL;
95 T *t = top->Curr; // this object is to be returned
96 PNextList<T> *t2 = top; // store top to delete
97 top = top->Next;
98 t2->Next = NULL; // unlink
99 delete t2;
100 Count--;
101 return t;
102 }
103
104 // Clone doubles the whole stack. Either an empty stack is
105 // used (giving UseThis as an argument) or a new stack is
106 // produced.
107 PStack<T> *Clone(PStack<T> *UseThis = NULL) {
108 PStack<T> *t1;
109 if (UseThis)
110 t1 = UseThis;
111 else
112 t1 = new PStack();
113 if (!top)
114 return t1; // return an empty stack
115 t1->Push(top->Curr); // push top node object
116 PNextList<T> *t2 = top;
117 while (t2->Next) { // add all node in t2 to t1
118 t2 = t2->Next;
119 t1->top->Add(t2->Curr);
120 }
121 t1->Count = Count; // synchronize Count
122 return t1;
123 }
124};
125// end of template PStack

127// class PReactionList:
128// implements two connected stacks of PChannel to organize the
129// already processed decay channels and the work to do.
130// The variable ReactionWeight contains the overall weight of
131// this specific channel.
132class PReactionList {
133 public:
134 PStack<PChannel> *Finished;
135 PStack<PChannel> *ToDo;
136 Double_t ReactionWeight;
137 Int_t ID;
138 static Int_t maxID;
139
140 PReactionList() {
141 Finished = new PStack<PChannel>;
142 ToDo = new PStack<PChannel>;
143 ReactionWeight = 1.;
144 ID = ++maxID;
145 };
146
147 ~PReactionList() {
148 delete Finished;
149 delete ToDo;
150 }
151
152 PReactionList *Clone() {
153 PReactionList *newRL = new PReactionList;
154 Finished->Clone(newRL->Finished);
155 ToDo->Clone(newRL->ToDo);
156 newRL->ReactionWeight = ReactionWeight;
157 return newRL;
158 }
159};
