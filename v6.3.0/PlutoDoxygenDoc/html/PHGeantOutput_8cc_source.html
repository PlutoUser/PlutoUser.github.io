2// File output interface implementation file
3//
4// This class serves as ASCII file output class.
5// This IO class supports all standard ascii io formats
6// of PReaction plus additional formats exclusivly used
7// by HGeant of HADES. The output flags getVERTEX and writeINDEX
8// are transported from the settings of PReaction.
9// Below the coding of the header flag for the Geant IO are listed.
10//
11// getVERTEX -> write out the vertex coordinates of the particles
12// writeINDEX -> write out parent index
13// writeSEQNUMBER -> write out event sequence number for each particle (used for embedding)
14//
15// //----------------------------------------------
16// // construct the header flag : normal flag + 10
17// Int_t shift = 0;
18// if(writeSEQNUMBER) shift = 10;
19// if(!getVERTEX && writeINDEX == 0) flag = (2 + shift);
20// if(!getVERTEX && writeINDEX != 0) flag = - (2 + shift);
21// if( getVERTEX && writeINDEX == 0) flag = (4 + shift);
22// if( getVERTEX && writeINDEX != 0) flag = - (4 + shift);
23// //----------------------------------------------
24//
25// The above header flags are analyzed by HGeant during ascii file io
26// to set the input mode.
27//
28//
29//----------------------------------------------
30// USAGE:
31//
32// {
33// PReaction my_reaction(..settings...); // create reaction
34//
35// ... // add all channels , bulk etc
36//
37// PHGeantOutput* output = new PHGeantOutput(); // created output
38// output->OpenFile("filename"); // open file
39// my_reaction.AddFileOutput(output); // connect output to raection
40//
41// ... // do all other actions
42//
43// my_reaction.Loop(nEvents); // run event loop
44// output->CloseFile(); // close output
45// }
46//
47// Author: Jochen Markert
48// Written: 05/12/2007
49// Revised:
50//

109bool PHGeantOutput::WriteEventHeader(void) {
110 // Write the event header. The io flag is coded
111 // by the diffent switches. The blast parameter
112 // is taken from the first channel (if it exists)
113 // otherwise blast will be 0.
114
115 Int_t flag = 0;
116 Float_t blast = 0;
117
118 if(channel) {
119 blast = channel[0]->GetBT();
120 }
121
122 //----------------------------------------------
123 // construct the header flag : normal flag + 10
124 Int_t shift = 0;
125 if(writeSEQNUMBER) shift = 10;
126 if(!getVERTEX && writeINDEX == 0) flag = (2 + shift);
127 if(!getVERTEX && writeINDEX != 0) flag = - (2 + shift);
128 if( getVERTEX && writeINDEX == 0) flag = (4 + shift);
129 if( getVERTEX && writeINDEX != 0) flag = - (4 + shift);
130 //----------------------------------------------
131
132
133 if(asciiFile){
134 fprintf(asciiFile, " %i %i %f %f %i\n", ctEvt, cnt, blast, *event_impact_param, flag);
135
136 ctEvt ++;
137 return kTRUE;
138 }
139
140 return kFALSE;
141}
