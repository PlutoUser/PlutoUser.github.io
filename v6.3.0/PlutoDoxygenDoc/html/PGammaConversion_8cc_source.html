98Float_t PGammaConversion::ConversionXS(Float_t E) { // return total conversion cross section in mb/atom (Geant3)
99
100 const Float_t EMass = 5.109990615e-4; // electron mass in GeV
101 if ( E<2.*EMass ) return 0; // unphysical region
102 Float_t Eph = (E<10 ? E : 10.);
103
104 // Coefficients from Geant3 routine GPRSGA
105 const Float_t CPar[18] = {0.87842E-3, -0.19625E-2, 0.12949E-2, -0.20028E-3, 0.12575E-4, -0.28333E-6,
106 -0.10342E-4, 0.17692E-4, -0.82391E-5, 0.13063E-5, -0.90815E-7, 0.23586E-8,
107 -0.45263E-3, 0.11161E-2, -0.86749E-3, 0.21773E-3, -0.20467E-4, 0.65372E-6};
108 Float_t CC[3] = {0, 0, 0};
109 Float_t X = log(Eph/EMass);
110
111 // initialising parameters of CC1, CC2, CC3
112 for (Int_t i=0; i<3; i++) {
113 Float_t D = 1;
114 for(int j=0; j<6; j++){
115 Int_t jx = 6*i + j;
116 CC[i] = CC[i] + D*CPar[jx];
117 D = D*X;
118 }
119 }
120
121 // total cross section for element with Z
122 return (Z*(Z+1.0) * (CC[0] + CC[1]*Z + CC[2]/Z));
123}

126Float_t PGammaConversion::GetConversionProb(Float_t E, Float_t th, Int_t run = 0) {
127 // return conversion prob for photon of energy E [GeV] and polar angle theta [deg]
128 if (th<0. || th>90)
129 return 0; // out of HADES acceptance
130 if (ConvProb >= 1.0)
131 return 1; // convert all
132 if (ConvProb > 0)
133 return ConvProb; // use constant conversion probability
134 else {
135 if (run != 2)
136 return 0.01 * convpar[0] * (1. + convpar[1]*th + convpar[2]*th*th) * ConversionXS(E);
137 if (run == 2)
138 return 0.01 * convpar[0] * (1. + convpar[1]*th + convpar[2]*th*th +
139 convpar[3]*th*th*th+convpar[4]*th*th*th*th) * ConversionXS(E);
140 }
141 return 0.;
142}

144Double_t PGammaConversion::dSde(Double_t *x, Double_t *par) { // electron energy distribution (GEANT3)
145 Double_t E = par[0]; // photon energy in GeV
146 Double_t Z = par[1]; // average atomic charge of converter material
147
148 Double_t eps = x[0]/E; // electron fraction of photon energy
149 if (eps<=0 || eps>=1 || Z<1)
150 return 0;
151
152 const Double_t alpha = 1./137.036; // fine structure constant
153 const Double_t me = 0.0005109989; // electron mass in GeV
154
155 Double_t a = pow(alpha*Z,2);
156 Double_t fcZ = alpha*(1/(1+alpha) + 0.20206 - 0.0369*a + 0.0083*a*a - 0.002*a*a*a); // Coulomb correction
157 Double_t FZ = E<0.05 ? 4./3.*log(Z) : 4./3.*log(Z)+4.*fcZ;
158
159 Double_t delta = 136.*me/(pow(Z,1./3.)*E*eps*(1-eps));
160 Double_t phi1, phi2; // screening functions
161 if (delta<=1) {
162 phi1 = 20.867 - 3.242*delta + 0.625*delta*delta;
163 phi2 = 20.209 - 1.930*delta - 0.086*delta*delta;
164 } else {
165 phi1 = phi2 = 21.12 - 4.184*log(delta+0.952);
166 }
167
168 phi1 = max(phi1, FZ); // to avoid negative result for eps close to 0 or 1
169 phi2 = max(phi2, FZ);
170
171 Double_t xiZ = log(1440./pow(Z,2./3.))/(log(183./pow(Z,1./3.)) - fcZ);
172
173 Double_t val = alpha*Z*(1+xiZ)/(E*E) * ((eps*eps+(1-eps)*(1-eps))*(phi1-FZ)
174 + 2./3.*eps*(1-eps)*(phi2-FZ));
175 return val;
176}

274Bool_t PGammaConversion::Modify(PParticle **mstack, int *decay_done, int *num, int stacksize) {
275 // Read the particles from the defined stack and copy this to the official
276 // particle stream
277
278 const Double_t R2D = 57.295780; // rad to deg
279 const Double_t me = 0.0005109989; // electron mass in GeV
280
281 int cur = *num;
282 int gamma_id = makeStaticData()->GetParticleID("g");
283
284 // cout << cur << " " << mstack[0]->ID() << endl;
285
286 for (int i=0; i<cur; i++) {
287
288 if (mstack[i]->HasID(gamma_id) && mstack[i]->IsActive()) { // this is a photon
289
290 if ((*num) == stacksize) {
291 Warning("Modify", "stacksize reached");
292 return kFALSE;
293 }
294
295 //make kinematics here....
296 PParticle *track = mstack[i];
297 TVector3 vertex = track->GetVertex();
298
299 Double_t Ephot = track->E();
300 Double_t thphot = R2D*track->Theta();
301
302 if ( thphot<0. || thphot>90 ) continue;
303 /*
304 Double_t convprob = GetConversionProb(Ephot, thphot); // get conversion probability
305 if ( RGen->Uniform()>convprob ) { // do not convert this one
306 decay_done[i] = 0; // do not suppress unconverted photons from output
307 continue;
308 }
309 */
310 // Int_t mesonId = track->GetParentId();
311
312 if (Ephot < 2*me)
313 return kFALSE;
314 Int_t index = int(Ephot/0.01); // select energy slice of 0.01 GeV/c
315 if (index < 0)
316 return kFALSE;
317 index = min(index, NEFUNCS-1);
318
319 if (Elepton[index] == NULL) { // this slice does not yet exist
320 Char_t funame[11] = "Elepton000";
321 sprintf(funame, "Elepton%03d", index); // create function object
322 Elepton[index] = new TF1(funame, PGammaConversion::dSde, me, double(index)*0.01+0.005, 2);
323 Elepton[index]->SetNpx(200);
324 Elepton[index]->SetParameters(Ephot, Z); // and initialize it
325 }
326
327 // Do energy and angle sampling of the e+/e- pair
328
329 Double_t Eelec = Elepton[index]->GetRandom(); // lepton total energy
330 Eelec *= Ephot/(double(index)*0.01+0.005); // correct for discretization of sampling
331 Double_t Pelec = sqrt(Eelec*Eelec - me*me); // lepton momentum
332 Double_t Thelec = (me/Ephot) * ulep->GetRandom(); // lepton polar angle (w.r.t. photon)
333 Double_t Eposi = Ephot - Eelec; // assume energy conservation
334 Double_t Pposi = sqrt(Eposi*Eposi - me*me);
335 Double_t arg = (Pelec/Pposi) * sin(Thelec); // assume coplanar photon e+ e-
336 if (arg<-1.) arg = -1;
337 if (arg> 1.) arg = 1;
338 Double_t Thposi = TMath::ASin(arg);
339
340 Double_t Phelec = 6.28318531 * RGen->Uniform(); // sample uniform phi angle
341 Double_t Phposi = 6.28318531 - Phelec; // assume that pair momentum is colinear
342 // Double_t close = R2D * (Thelec + Thposi); // opening angle of conversion pair
343 TVector3 Dirphot(track->Vect()); // photon direction
344 Dirphot = Dirphot.Unit(); // normalize photon vector
345
346 TVector3 lep3(Pelec*cos(Phelec)*sin(Thelec), Pelec*sin(Phelec)*sin(Thelec), Pelec*cos(Thelec));
347 TVector3 lep4(Pposi*cos(Phposi)*sin(Thposi), Pposi*sin(Phposi)*sin(Thposi), Pposi*cos(Thposi));
348 lep3.RotateUz(Dirphot); // rotate lepton vectors into lab frame
349 lep4.RotateUz(Dirphot);
350
351 /*
352 Float_t ptot3 = lep3.Mag();
353 Float_t theta3 = R2D*lep3.Theta();
354 Float_t phi3 = R2D*TMath::ACos(lep3.X()/lep3.Perp());
355 if (lep3.Y() < 0.) phi3 = 360.-phi3;
356 Float_t ptot4 = lep4.Mag();
357 Float_t theta4 = R2D*lep4.Theta();
358 Float_t phi4 = R2D*TMath::ACos(lep4.X()/lep4.Perp());
359 if (lep4.Y() < 0.) phi4 = 360.-phi4;
360 */
361
362 Int_t id3, id4;
363 if (RGen->Uniform() < 0.5) { // decide which is e- and which is e+
364 id3 = 3;
365 id4 = 2;
366 } else {
367 id3 = 2;
368 id4 = 3;
369 }
370
371
372 ep.SetVectM(lep3,me);
373 ep.SetID(id3);
374 ep.SetSourceId(track->GetSourceId());
375 ep.SetParentId(1);
376 ep.SetParentIndex(track->GetIndex());
377 ep.SetW(track->W());
378 ep.SetVertex(0, 0, 0, 0);
379 ep.SetActive();
380
381 em.SetVectM(lep4,me);
382 em.SetID(id4);
383 em.SetSourceId(track->GetSourceId());
384 em.SetParentId(1);
385 em.SetParentIndex(track->GetIndex());
386 em.SetW(track->W());
387 em.SetVertex(0, 0, 0, 0);
388 em.SetActive();
389
390
391 if (nVertex > 0) {
392 //----------------------------------
393 // Convert vertex into target num
394 TH2F *hzr = getHist(vertex.Z(), thphot);
395
396 if(!hzr) {
397 Error("Modify()", "hzr at theta %f and vertex %f not found !", thphot, vertex.Z());
398 continue;
399 }
400
401 Double_t z,r;
402 hzr->GetRandom2(z, r);
403
404 if(!TMath::Finite(z) || !TMath::Finite(r)) {
405 ep.SetVertex(0, 0, 10000, 0);
406 em.SetVertex(0, 0, 10000, 0);
407 } else {
408 Double_t phi = gRandom->Rndm()*2*TMath::Pi();
409 Double_t x = cos(phi)*r;
410 Double_t y = sin(phi)*r;;
411 ep.SetVertex(x, y, z, 0);
412 em.SetVertex(x, y, z, 0);
413
414 cout<<" x " << x << " y " << y << " phi " << phi*TMath::RadToDeg() <<
415 " r " << r << " th " << thphot << endl;
416 }
417 } else {
418 ep.SetVertex(x1, y1, z1, 0);
419 em.SetVertex(x1, y1, z1, 0);
420 }
421
422 *(mstack[*num]) = ep;
423 (*num)++;
424 *(mstack[*num]) = em;
425 (*num)++;
426 decay_done[i] = 1;
427 }
428 }
429
430 return kTRUE;
