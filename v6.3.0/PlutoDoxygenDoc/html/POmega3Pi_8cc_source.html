45Bool_t POmega3Pi::Init(void) {
46
47 //looking for primary. This is mandatory
48 primary = GetParticle("primary");
49 if (!primary) {
50 Warning("Init", "Primary not found");
51 return kFALSE;
52 }
53
54 //now get the parent
55 for (int i=0; i<position; i++) {
56 if (particle_flag[i] == PARTICLE_LIST_PARENT)
57 parent = particle[i];
58 }
59 if (!parent) {
60 Warning("Init", "Parent not found");
61 return kFALSE;
62 }
63
64 int side = 0;
65
66 for (int i=0; i<position; i++) {
67 if ((particle_flag[i] & PARTICLE_LIST_DAUGHTER)
68 && (particle[i] != primary)) {
69 if (side == 2) {
70 Warning("Init", "More than 2 side particle found");
71 return kFALSE;
72 }
73 side_particle[side] = particle[i];
74 side++;
75 }
76 }
77
78 if (side != 2) {
79 Warning("Init", "Less then 2 side particle found");
80 return kFALSE;
81 }
82
83 return kTRUE;
84};

98Bool_t POmega3Pi::IsNotRejected(void) {
99
100 TLorentzVector M00 = (*(TLorentzVector *) primary) + (*(TLorentzVector *) side_particle[0]);
101 TLorentzVector M01 = (*(TLorentzVector *) primary) + (*(TLorentzVector *) side_particle[1]); // Definiere meinen TLorentzvector für pi0/pi- auf x-Achse
102
103 double myvariable = M01.M2();
104 double myvariable2 = M00.M2();
105
106 double factor = POmega3Pi::diffgam(myvariable, myvariable2);
107
108
109 if (factor > max) {
110 Warning("IsNotRejected", "Dalitz factor %f > max %f", factor, max);
111 if (max < 0) {
112 Warning("IsNotRejected", "Dalitz factor max not set");
113 }
114 max = factor*1.02;
115 }
116
117 if (factor < 0) {
118 return kFALSE;
119 }
120
121 if (((factor/max)) > PUtils::sampleFlat()) {
122 return kTRUE;
123 }
124
125 return kFALSE;
126};

129double POmega3Pi::diffgam(double M00, double M01) {
130 //Written by S. Leupold
131 double pi = TMath::Pi();
132
133 // physical parameters m_V, h_P, h_A, f, pion mass m_pi, omega mass m_omega
134 double mv, hp, ha, f, mpi, mom;
135 // third combination of invariant masses (squared)
136 double M02;
137 // matrix element C_omega -> 3pi
138 TComplex cc;
139 // parts of cc
140 TComplex h1, h2, h3, addh;
141 // phase space, returned variable
142 double p, dg;
143
144 TComplex a,b,c,d,e,ff;
145
146 //Take Pluto build-in rho propagator
147 if (RhoPropagator == NULL) {
148 RhoPropagator = makeDynamicData()->GetParticleSecondaryModel("rho0", "propagator");
149 if (RhoPropagator == NULL)
150 Fatal ("diffgam", "RhoPropagator not defined");
151 }
152
153
154 mv = 0.78;
155 hp = 0.304;
156 ha = 2.1;
157 f = 0.09;
158 mpi = 0.14;
159 mom = 0.78;
160
161 M02 = -M00 - M01 + TMath::Power(mom,2) + 3*TMath::Power(mpi,2);
162
163
164 p = -1./3.*((TMath::Power(M00,2)*M01)/4. + (M00*TMath::Power(M01,2))/4.
165 - (M00*M01*TMath::Power(mom,2))/4. -
166 (3*M00*M01*TMath::Power(mpi,2))/4. + (TMath::Power(mom,4)*TMath::Power(mpi,2))/4. -
167 (TMath::Power(mom,2)*TMath::Power(mpi,4))/2. + TMath::Power(mpi,6)/4.);
168
169
170
171 a = M00 + TMath::Power(mom,2);
172 h1 = RhoPropagator->GetAmplitude(&M00)*a.Re();
173
174 b = M01 + TMath::Power(mom,2);
175 h2 = RhoPropagator->GetAmplitude(&M01)*b.Re();
176
177 c = M02 + TMath::Power(mom,2);
178 h3 = RhoPropagator->GetAmplitude(&M02)*c.Re();
179
180 d = h1+h2;
181 addh = d+h3;
182
183 e = ha*hp*mv / (4.*TMath::Power(f,3)*mom);
184 cc = addh * e;
185
186 ff = p*cc*TComplex::Conjugate(cc);
187 dg = ff/(8.*pi*pi*pi*32.*mom*mom*mom);
188
189 return dg;
190};
