81Double_t PEtaPiPiDilepton::GetWeight(Double_t *mass, Int_t *) {
82 //Inputs: s_pipi, q, theta_pip, theta_e, phi
83 //This is the second part of eqn. 64 of Wirzba's report
84
85 Double_t s_pipi = mass[0]*mass[0];
86 Double_t q = mass[1];
87 Double_t q2 = q*q;
88 Double_t theta_pip = mass[2];
89 Double_t theta_ep = mass[4];
90 Double_t phi = mass[5];
91
92 Double_t sin_theta_pip = sin(theta_pip);
93 Double_t beta_pi2 = (1-((4*m_pi*m_pi)/s_pipi));
94 Double_t weight = (s_pipi*beta_pi2 *sin_theta_pip*sin_theta_pip);
95
96 //for the curly bracket only the first part is supported at the moment
97 Double_t beta_e2 = (1-((4*mass_e*mass_e)/q2));
98 Double_t weight_part2 = PKinematics::lambda(mass_eta*mass_eta,s_pipi,q2) *
99 (1+beta_e2*sin(theta_ep)*sin(theta_ep) * sin(phi)*sin(phi));
100
101 return weight * weight_part2;
102}

108Double_t PEtaPiPiDilepton::GetWeight(void) {
109 //Returns the weight of ref [L1]
110
111 TLorentzVector twopi = ((*(TLorentzVector*) pip)
112 + (*(TLorentzVector*) pim));
113 TLorentzVector dil = ((*(TLorentzVector*) ep)
114 + (*(TLorentzVector*) em));
115
116
117 PParticle p_star(pip);
118 //rotate such that twopi points to z-axis:
119 Double_t Phi = twopi.Phi();
120 Double_t Theta = twopi.Theta();
121 twopi.RotateZ(-Phi);
122 twopi.RotateY(-Theta);
123 p_star.RotateZ(-Phi);
124 p_star.RotateY(-Theta);
125 p_star.Boost(-twopi.BoostVector());
126
127 Double_t theta_pi = TMath::Pi() - p_star.Theta();
128
129 Double_t s_pipi = twopi.M();
130
131 PParticle ep_tmp(pip);
132 Phi = dil.Phi();
133 Theta = dil.Theta();
134 dil.RotateZ(-Phi);
135 dil.RotateY(-Theta);
136 ep_tmp.RotateZ(-Phi);
137 ep_tmp.RotateY(-Theta);
138 ep_tmp.Boost(-dil.BoostVector());
139
140 Double_t phi=p_star.Phi() - ep_tmp.Phi();
141
142 return GetWeight(s_pipi, dil.M(), theta_pi, ep_tmp.Theta(), phi);
143}

145Bool_t PEtaPiPiDilepton::IsNotRejected(void) {
146 //Use rejection mode...
147
148 if (GetVersionFlag(VERSION_WEIGHTING)) return kTRUE;
149 //...but not if weighting enabled.
150
151 Double_t weight = GetWeight();
152
153 if (weight>weight_max) {
154 Warning("IsNotRejected", "Weight (%lf) > max (%lf)", weight, weight_max);
155 weight_max = weight*1.1;
156 }
157
158 if ((weight/weight_max)>PUtils::sampleFlat())
159 return kTRUE; // sample now distribution
160
161 return kFALSE;
162}
