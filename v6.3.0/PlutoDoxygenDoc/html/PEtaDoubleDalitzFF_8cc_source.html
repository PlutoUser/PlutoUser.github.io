49Double_t PEtaDoubleDalitzFF::GetWeight() {
50
51 TLorentzVector *P3, *P4;
52 P3 = (TLorentzVector*) dil1;
53 P4 = (TLorentzVector*) dil2;
54
55
56 //******************* Extraction of physics observables ****************
57 Double_t q1_2 = (*P3)*(*P3);//Lorentz Vector of 1st g squared
58 Double_t q2_2 = (*P4)*(*P4);//Lorentz Vector of 2nd g squared
59 //*************FF from Johan and Fredrik paper***********************
60 // Double_t FF=Lambda^4/(Lambda^2+mVV[0])*Lambda/(Lambda^2+mVV[1]);// including process with Lambda parameter - Ro meson
61 // Double_t FF=Lambda*Lambda*Lambda*Lambda/((Lambda*Lambda-q1_2)*(Lambda*Lambda-q2_2));
62
63 // Double_t FF=Lambda*Lambda/((Lambda*Lambda-q1_2));
64 Double_t FF = Lambda*Lambda*Lambda*Lambda/((Lambda*Lambda-q1_2)*(Lambda*Lambda-q2_2));
65 float weightFF = FF;
66
67 //weightFF = 20;
68 // cout << weightFF << endl;
69 return weightFF;
70}
