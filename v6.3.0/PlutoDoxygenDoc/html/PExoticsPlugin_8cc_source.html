137 makeStaticData()->AddAlias("phi2170", "phi(2170)");
138 makeStaticData()->SetParticleCharge("phi2170", 0);

140 makeStaticData()->SetParticleMeson("phi2170", kTRUE);

142 makeStaticData()->AddDecay("phi(2170) --> phi + pi+ + pi-", "phi2170", "phi,pi+,pi-", 1.0/3.0);

145 PHadronDecayM2Mesh *model = new PHadronDecayM2Mesh("phi2170_m2mesh_phi_f0_980", "Mesh decay for phi2170", decay_key);
146 pdmutil->Add(model);
147 } else {
148 Error("ExecCommand", "PID blocked for phi2170, plugin disabled");
149 (*system_alloc_verbosity) = old;

151 }
152
153 (*system_alloc_verbosity) = old;
154 }
155 return kTRUE;
156 }
157
158 return kFALSE;
159}
