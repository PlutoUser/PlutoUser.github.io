66Double_t PDeltaDalitzFF::facteur_rho(Double_t x) {
67 // rho propagator term
68
69 //Iachello "old" version
70 if ((x < fm_pis) && (useQED != -2)) return mass_rho*mass_rho
71 /(mass_rho*mass_rho - x);
72 //scaling to avoid "step"
73 // Double_t scale=0.91;
74 Double_t scale = 1.;
75 Double_t alpha = 1.;
76 Double_t beta = 0.;
77
78 if (useQED == -1)
79 gamma_rho = makeDynamicData()->GetParticleTotalWidth(sqrt(x), rhopid);
80
81 if ((useQED == -2) && (x < fm_pis)) { //TODO: Make a more meaningfull flag name
82 alpha = sqrt((fm_pis - x)/x)*(1.-2/TMath::Pi() *atan(sqrt((fm_pis - x)/x)));
83 beta = 0.;
84 } else {
85 Double_t qnorm = x/fm_pis;
86 Double_t logpart = (sqrt(x - fm_pis ) + sqrt(x))/(2*mass_pi);
87 alpha = 2/TMath::Pi()*sqrt((x - fm_pis )/x)*log(logpart);
88 beta = sqrt(pow(qnorm-1.,3)/qnorm);
89 }
90
91 // Double_t logpart = (sqrt(x - fm_pis ) + sqrt(x))/(2*mass_pi);
92 // Double_t alpha= 2/TMath::Pi()*sqrt((x - fm_pis )/x)*log(logpart);
93 Double_t denom_reel = mass_rho*mass_rho - x + (fm_pis -x)*gamma_rho*alpha/mass_pi;
94 // Double_t qnorm=x/fm_pis;
95 // Double_t beta = sqrt(pow(qnorm-1.,3)/qnorm);
96 double denom_im = gamma_rho*4*mass_pi*beta;
97 return scale*(mass_rho*mass_rho +
98 8*gamma_rho*mass_pi/TMath::Pi())
99 /sqrt(denom_reel*denom_reel+denom_im*denom_im);
100}

102Double_t PDeltaDalitzFF::GetWeight(Double_t *mass, Int_t *) {
103
104 Double_t x = mass[0]*mass[0];
105 Double_t m_d2 = mass[1]*mass[1]; //mass of Delta sq.
106
107 //q-dependent QED-Formfactor
108 if (useQED == 1)
109 return (g_m2 + 3 * g_e2 + (g_c2*x/(2*m_d2)));
110
111 //VMD-Formfactor
112 Double_t ff = 2.4637*gk2(-x)*(beta_prime + beta*facteur_rho(x));
113 if (rescale) ff *= sqrt((4.76826/4.68927)*(g_m2/(3.2*3.2)));
114 return ff*ff;
115
116}
