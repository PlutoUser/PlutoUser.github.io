2// Read in urqmd f15 file (collision history) and put the particles
3// on the PLUTO bulk stack. Either dileptons (D0,D+,w,rho0,phi,eta,eta'
4// and pi0) can be decayed via shining (translated for original UrQMD
5// fortran routines, authors: Katharina Schmidt, Sascha Vogel,
6// Christian Sturm 2007-2009) into e+/e- or by PLUTO bulk decay.
7// Optional stable non dileptons can be written out. PHUrReader adds
8// a TClonesArray "Addon" to the PLUTO Tree that contains original
9// UrQMD infos about creation and absorbtion times, densities etc.
10//
11// USAGE: see urqmd_f15_input.C

43 Taken from Unigen:
44 UrQMD PIDs are in fact composite - a particle is fully defined by the
45 type specifier (ityp), the charge (ichg) and in case of baryons, the
46 third component isospin (iso3; ignored by us at present). For
47 simplicity, our conversion tables collapse these into a single number
48 as follows:
49 - propagate the sign of ityp (particle-antiparticle distinction for
50 baryons, strangeness-antistrangeness distinction for mesons) to that
51 of the stored value;
52 - shift the ichg range from -2..2 (UrQMD does not support nuclear
53 fragments other than protons and neutrons so all particles it
54 produces fall in this range) to 0..4 to make sure it doesn't
55 interfere with the above;
56 - multiply shifted charge by +/-1000 and add it to type. The latter
57 is guaranteed to be smaller than 1000 (baryon types are one- or
58 two-digit, meson types three-digit) so that way no ambiguities
59 occur.
60 Int_t id;
61 if (ityp >= 0)
62 id = 1000 * (ichg + 2) + ityp;
63 else
64 id = -1000 * (ichg + 2) + ityp;
65 */

470Bool_t PHUrReader::ReadEvent() {
471 // read function for UrQMD f15 files
472
473 if(!fInputAscii.is_open() || !fInputAscii.good()) return kFALSE;
474
475 if(fEvtCt == 0){
476 // only in first event. In next
477 // events the eventheader will be read
478 // by the previous call.
479 evtheader.Clear(NULL);
480 if(!evtheader.Read(fInputAscii)){
481 Error("ReadEvent()", "Could not read event header!");
482 return kFALSE;
483 }
484 } else {
485 evtheader = evtheaderCP;
486 }
487
488 if(evtheader.id == -1 || evtheader.id == 0) { // detected eventheader
489
490 if(printEvtHeader) { evtheader.Print(); }
491
492 ClearVector(collisionsIn); // clear collisions of previous event
493 ClearVector(collisionsOut);
494 collisions.clear();
495
496 while ( !fInputAscii.eof() ) {
497 colheader.Clear(NULL);
498 if(!colheader.Read(fInputAscii)) return kFALSE; // read next collision header
499
