25Double_t PEtaDoubleDalitz::Gen2lepton1(Double_t m) {
26 Double_t MEl = PData::LMass("e+");
27 Double_t MEl2 = 2*MEl;
28 Double_t m2 = m*m;
29 Double_t ARAND[2];
30 Double_t WTD;
31 Double_t AM;
32 do {
33 gRand->RndmArray(2, ARAND);
34 AM = MEl2*TMath::Exp(TMath::Log(m/MEl2)*ARAND[0]);
35 Double_t AM2 = AM*AM;
36 Double_t RE = MEl2*MEl2/AM2;
37 WTD = TMath::Sqrt(1-RE)*(1+RE/2)*TMath::Power(1-AM2/m2,3);
38 } while(ARAND[1]>WTD); // ???????????? while(ARAND[1]>WTD)
39 return AM;
40}

72Bool_t PEtaDoubleDalitz::SampleMass(void) {
73
74 Double_t MEta = parent->M();
75
76 Double_t weight = 1.;
77 Double_t mVV[2];
78 do {
79 mVV[0] = Gen2lepton1(MEta);//IM of the 1-st pair e+e-
80 mVV[1] = Gen2lepton1(MEta);//IM of the 2-nd pair e+e-
81
82 if (formfactor_model && formfactor_model->GetVersionFlag(VERSION_SAMPLING)) {
83 weight *= formfactor_model->GetWeight(mVV[0]);
84 weight *= formfactor_model->GetWeight(mVV[1]);
85 }
86
87 } while((mVV[0]+mVV[1]>MEta) || ((weight/ff_w_max) < PUtils::sampleFlat()) );
88
89
90 dil1->SetM(mVV[0]);
91 dil2->SetM(mVV[1]);
92
93 return kTRUE;
