2// Pluto SAID parametrization implementation file
3// Low energy SAID pp-elastic parametrization.
4// Created by Nodar Lomidze and Mirian Tabidze;
5// ANKE/PAX Collaboration 03.09.09
6//
7// For each theta (0-180) deg. dsg/dT_kin distributions was fitted using function
8// a*pow[T_kin,b*pow(T_kin,c)] + d, where a,b,c, and d parameters are functions of
9// theta, T_kin is proton kinetic energy.
10// For approximation this parameters full theta region was divided into 5 intervals
11// (0-1), (1-4), (4-20), (20-40) and (40-90) deg.
12// In each interval polynomial functions (pol4-pol6) or function
13// pol1+A*x^B*sin^C(x)*exp(D*x) was used (see code below),
14// where A,B,C and D are parameters.
15// In (90-180) deg. interval same function was used as in (90-0) deg.
16//
17// Differences from SAID above 4 MeV incident proton kinetic energy in theta
18// (25-90) deg. are less then 1%, in (4-25) deg. ~4-6% and in (1-4) deg. (10-15)%.
19// Below 4 MeV parametrization is worse ~(10-20)% in all theta intervals.
20//
21//

242double PSaidLowEnergy::sample(double r, double tlab) {
243 const double fac = 1.3;
244 int i, j;
245 // _____________________________________________________________________
246 // enter whenever beam energy changes, to calculate integrals and slopes
247 if (told != tlab) {
248 told = tlab;
249 double sum = 0.;
250 for (i=0; i<=dim; ++i)
251 y[i] = fac * dsdw((double)(i+1), tlab);
252 for (i=0; i<dim; ++i) {
253 sum += (y[i+1]>y[i]) ? y[i+1] : y[i] ;
254 aa[i] = sum;
255 }
256 }
257
258 // _____________________________________________________________________
259 // sample cm scattering angle with the rejection method
260 double rr, f, tf, r2, c0, r1 = r;
261
262 repeat:
263 i = (r1>.5);
264 rr = (2.*r1-i)*aa[dim-1];
265 for (j=0; j<dim; ++j) {
266 if (rr<aa[j]) break;
267 }
268 if (j) rr -= aa[j-1];
269 tf = (y[j+1]>y[j]) ? y[j+1]: y[j];
270 c0 = rr/tf+1.+j;
271 f = dsdw(c0,tlab);
272 //r2=REngine->Rndm();
273 r2 = PUtils::sampleFlat();
274 if (r2 > f/tf) goto fail;
275 if (f <= tf) return (1-2*i)*cos(c0*d2r);
276
277 Warning("sample", "sampling failed f=%lf, tf=%lf", f, tf);
278 exit(1);
279
280 fail:
281 //r1=REngine->Rndm();
282 r1 = PUtils::sampleFlat();
283 goto repeat;
284}
