32Double_t PBremsstrahlung::DLines(Double_t x, Double_t par0, Double_t par1, Double_t par2, Double_t par3) const {
33 Double_t m,n;
34
35 if(x < 1.9) {
36 m = (par1-par0)/(0.65);
37 n = par0-m*1.25;
38 return m*x+n;
39 } else {
40 if(x < 2.2) {
41 m = (par2-par1)/(0.3);
42 n = par1-m*1.9;
43 return m*x+n;
44 } else {
45 m = (par3-par2)/(1.3);
46 n = par2-m*2.2;
47 return m*x+n;

101 threshold = 2*makeStaticData()->GetParticleMass("p");

104 }

118void PBremsstrahlung::SetMode(Char_t mode) {
119 // model [0..4] where 0 => bare gNN Bremsstrahlung [default]
120 // 1 => gDN Delta contribution included
121 // 2 => cSum coherent sum of bare an gDN Bremsstrahlung
122 // 3 => N1520 contribution (Shyam/Mosel only)
123 // The following modes are untested (IF, 26.9.2008)
124 // 4 => FSI final state interactions included
125 // 5 => VMD Vector Meson Dominance contribution
126 // 6 => VMD and FSI contribution included
