39Bool_t PEtaPiPiGamma::Init(void) {
40 //Init function called once for each PChannel
41
42 //looking for parent. This is mandatory
43 parent = GetParticle("parent");
44 if (!parent) {
45 Warning("Init", "Parent not found");
46 return kFALSE;
47 }
48 pip = GetParticle("pi+");
49 if (!pip) {
50 Warning("Init", "Pi+ not found");
51 return kFALSE;
52 }
53 pim = GetParticle("pi-");
54 if (!pim) {
55 Warning("Init", "Pi- not found");
56 return kFALSE;
57 }
58 gamma = GetParticle("g");
59 if (!gamma) {
60 Warning("Init", "Gamma not found");
61 return kFALSE;
62 }
63
64 m_formfactor_model =
65 GetSecondaryModel("formfactor");
66
67 //Fold the 2 max values
68 if (m_formfactor_model) {
69 Info("Init", "Found FF model");
70 Double_t ff_w_max = m_formfactor_model->GetWeightMax();
71 if (ff_w_max < 0) {
72 Warning("Init", "Max value of the FF model not initialized");
73 } else {
74 weight_max *= ff_w_max;
75 }
76 }
77
78 return kTRUE;
79}

82Double_t PEtaPiPiGamma::GetWeight(void) {
83 //Returns the weight of ref [L1]
84
85 Double_t e_tilde = gamma->E();
86
87 TLorentzVector twopi = ((*(TLorentzVector*) pip)
88 + (*(TLorentzVector*) pim));
89
90 PParticle p_star(pip);
91 //rotate such that twopi points to z-axis:
92 Double_t Phi = twopi.Phi();
93 Double_t Theta = twopi.Theta();
94 twopi.RotateZ(-Phi);
95 twopi.RotateY(-Theta);
96 p_star.RotateZ(-Phi);
97 p_star.RotateY(-Theta);
98
99 p_star.Boost(-twopi.BoostVector());
100
101 Double_t theta_pi = TMath::Pi() - p_star.Theta();
102 Double_t p_star_value = p_star.P();
103 Double_t s_pipi = twopi.M(); //to be consistent I used the mass (not squared) - see SECONDARY_MODELS
104
105 Double_t weight = (
106 e_tilde*e_tilde * p_star_value*p_star_value
107 * pow(sin(theta_pi),2));
108
109 if (m_formfactor_model) {
110 weight *= m_formfactor_model->GetWeight(s_pipi);
111 }
112
113 return weight;
114
115}

117Bool_t PEtaPiPiGamma::IsNotRejected(void) {
118 //Use rejection mode...
119
120 if (GetVersionFlag() & VERSION_WEIGHTING) return kTRUE;
121 //...but not if weighting enabled.
122
123 Double_t weight = GetWeight();
124
125 if (weight > weight_max) {
126 weight_max = weight*1.1;
127 Warning("IsNotRejected", "Weight > max, new max is %lf", weight_max);
128 }
129
130 if ((weight/weight_max) > PUtils::sampleFlat())
131 return kTRUE; // sample now distribution
132
133 return kFALSE;
134}
