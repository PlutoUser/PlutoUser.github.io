1//TITLE <b>Virtual detectors and batch packs</b> This macro makes a demo filter root file
2
3{
4 //Let us create a "virtual detector"
5
6 //First, we create dummy root files
7 //For a real detector, they can be derived e.g.
8 //from a full GEANT simulation and re-opened
9 //here from a ROOT file
10
11 //Our dummy detector should have an acceptance
12 //for electrons in forward directions
13 //between 18 - 95 degrees. The efficiency
14 //of electrons should be 80%,
15 //in a momentum range below 100 MeV only 50,
16 //and below 50 MeV no acceptance
17
18 //Protons should have a flat acceptance of 90%
19 //using a forward detector (a plane) in
20 //a distance of 2m downstream with 3x3m
21
22 //Other particles are rejected
23
24 //Remember, this is NOT HADES
25
26 //This filter should come in 2 modi operandi:
27 //1. For inclusive measurements (e+e-)
28 //2. For exclusive measurements (ppe+e-)
29
30
31 //Make a pseudo-Monto-Carlo simulation:
32 //Generate particles between:
33 //mom = 0 ... 3000 MeV (x)
34 //cos(theta) = -1 ... 1 (y)
35 //phi = 0 ... 360 deg (z)
36
37 //On purpose, we do not use the Pluto convention
38 //to demonstrate the convertion inside the command list
39
40 //In addition, we apply a 1% momentum resolution...
41 //...just fixed, to keep it easy. Of course it is possible
42 //to use matrices as well
43 //This is controlled via "filter_smear_factor"
44
45 //Create the efficiency matrices:
46 TH3F *gen =
47 new TH3F ("gen", "Generated events", 30, 0, 1500, 20, -1, 1, 20, 0, 360);
48 TH3F *eff_elec =
49 new TH3F ("eff_elec", "Efficiency matrix electrons", 30, 0, 1500, 20, -1, 1, 20, 0, 360);
50 TH3F *eff_protons =
51 new TH3F ("eff_protons", "Efficiency matrix protons", 30, 0, 1500, 20, -1, 1, 20, 0, 360);
52
53 //Start the Monte-Carlo
54 int nev = 1000000;
55 for (int i=0; i<nev; i++) {
56 Double_t mom = PUtils::sampleFlat()*3000;
57 Double_t costheta = PUtils::sampleFlat()*2 - 1;
58 Double_t phi = PUtils::sampleFlat()*360;
59
60 gen->Fill(mom, costheta, phi);
61
62 Double_t theta = acos(costheta)*180/TMath::Pi();
63
64 //electrons
65 if ((theta > 18) && (theta < 95)) {
66 Double_t rnd = PUtils::sampleFlat();
67 if ((mom > 100) && (rnd < 0.8))
68 eff_elec->Fill(mom, costheta, phi);
69 else if ((mom > 50) && (rnd < 0.5))
70 eff_elec->Fill(mom, costheta, phi);
71 }
72
73 //protons
74 Double_t x = tan(acos(costheta)) * 2 * cos(phi*TMath::Pi()/180);
75 Double_t y = tan(acos(costheta)) * 2 * sin(phi*TMath::Pi()/180);
76
77 if ((fabs(x) < 1.5) && (fabs(y) < 1.5)) {
78 Double_t rnd = PUtils::sampleFlat();
79 if (rnd < 0.90)
80 eff_protons->Fill(mom,costheta,phi);
81 }
82 }
83
84 eff_elec->Divide(gen);
85 eff_protons->Divide(gen);
86
87
