32double PDeltaDalitzKrivoruchenko::dGdM(const int& id, const double& m, const double& ecm) {
33 //dGdM from Wolf expression:
34
35 double dgdm = 2.*alpha/(3.*TMath::Pi()*m);
36
37 if (m<mass_ee || m>ecm-mass_x) return 0.;
38// cout << m << ":" << ecm << endl;
39
40 double mn = mass_n;
41 if (id==Delta_plus || id==S11_plus)
42 mn = mass_p;
43
44 double ff = 2.7*2.7;
45
46 if (formfactor_model)
47 ff=formfactor_model->GetWeight(m,ecm);
48
49 dgdm *= (alpha/16.) *
50 (ecm + mn) * (ecm + mn) / (ecm*ecm*ecm * mn*mn) *
51 sqrt((ecm+mn) * (ecm+mn) - m*m) *
52 sqrt(
53 ((ecm-mn) * (ecm-mn) - m*m) *
54 ((ecm-mn) * (ecm-mn) - m*m) *
55 ((ecm-mn) * (ecm-mn) - m*m)
56 ) * ff;
57
58 // cout << dgdm << endl;
59
60 return dgdm;
61
62}
