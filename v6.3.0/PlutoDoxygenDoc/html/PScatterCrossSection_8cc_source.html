92Bool_t PScatterCrossSection::Prepare(void) {
93
94 PAngularDistribution::Prepare();
95
96 if (!pf2) {
97 Error("Prepare", "Function not defined");
98 return kFALSE;
99 }
100
101 pf2->GetRandom2(costheta, q);
102
103 Double_t m_t = target->M();
104 Double_t m_b = beam->M();
105
106 //momentum fixed target:
107 Double_t mom_b = sqrt( pow( (q*q-m_b*m_b-m_t*m_t)/(2*m_t) ,2) - m_b*m_b );
108
109 //cout << costheta << ":" << q << ":" << mom_b << endl;
110
111 if (std::isnan(mom_b)) {
112 return kFALSE;
113 }
114
115 //return kTRUE;
116 //Make a copy of beam and target:
117 PParticle my_beam(beam);
118 PParticle my_target(target);
119
120 my_beam.Boost(-my_target.BoostVector());
121 my_beam.SetMom(mom_b);
122 my_beam.Boost(my_target.BoostVector());
123
124 beam->SetVect(my_beam.Vect());
125 beam->SetM(m_b);
126
127 parent->Reconstruct();
128
129 return kTRUE;
130};
