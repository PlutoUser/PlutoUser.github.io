47double PHadesParticleSmearer::GetResolution(PParticle *pPart, Int_t iSetup) {
48
49 //double resolution_factor = 3;
50 double theta = TMath::RadToDeg()*(pPart->Theta()); // polar angle in degree
51 double phi = 90.-TMath::RadToDeg()*(pPart->Phi()); // azimuthal angle in degree
52 double mass = pPart->M()*1000.;
53 double p = (pPart->Vect()).Mag()*1000.;
54 if (phi<0.) phi += 360.; // TLorentzVector uses ATan2(x,y)
55 if(theta <= 15 || theta > 85) return 0.004;
56 Int_t thetaBin = int((theta-15)/10);
57 Int_t phiBin = int((phi - int(phi/60)*60)/12);
58 double c1, c2, res;
59 switch(iSetup) {
60 case 2:
61 c1 = detkick[thetaBin][phiBin];
62 res = c1*p*p;
63 break;
64
65 case 3:
66 c1 = detrk3[thetaBin][phiBin];
67 c2 = multscrk3[thetaBin][phiBin];
68 res = sqrt(c1*c1*p*p*p*p + c2*c2*(p*p + mass*mass));
69 break;
70
71 case 4:
72 c1 = detrk4[thetaBin][phiBin];
73 c2 = multscrk4[thetaBin][phiBin];
74 res = sqrt(c1*c1*p*p*p*p + c2*c2*(p*p + mass*mass));
75 break;
76
77 default:
78 res = 100;
79 }
80 res = resolution_factor*res/1000;
81 return res;
82}

84void PHadesParticleSmearer::Smear(PParticle *p, double gamma) {
85
86 //double ptot=(p->Vect()).Mag();
87 double ptot = p->P();
88 if (ptot <= 0.) return;
89 double th = p->Theta(), ph=p->Phi(), sth =sin(th), mass=p->M();
90 if (fabs(sth) < 1.e-6)
91 sth = (sth>0.) ? 1.e-6 : -1.e-6;
92 double smeared_ptot = ( rand.Gaus(ptot, gamma ) );
93 if(gamma == 0.1) smeared_ptot = ptot+0.04;
95 // smeared_ptot= ptot;
96 // cout<<ptot<<" "<< smeared_ptot <<endl;
98 //if(fabs(ptot-smeared_ptot)>0.15) cout << ptot << " " << smeared_ptot << endl;
99 double smeared_theta = rand.Gaus(th, 0.0023 );
100 double smeared_phi = ( rand.Gaus(sth*ph, 0.0023 ) )/sth;
101 double theta_check = smeared_theta-2.*TMath::Pi()*int(smeared_theta/(2.*TMath::Pi()));
102 if (theta_check>TMath::Pi() || theta_check<0.)
103 smeared_phi -= TMath::Pi();
104 p->SetRho(smeared_ptot); // keeping theta, phi constant
105 p->SetTheta(smeared_theta); // keeping |p|, phi constant
106 p->SetPhi(smeared_phi); // keeping |p|, theta constant
107 double newp=p->P();
108 p->SetE(sqrt(newp*newp+mass*mass)); // reset energy
109}
