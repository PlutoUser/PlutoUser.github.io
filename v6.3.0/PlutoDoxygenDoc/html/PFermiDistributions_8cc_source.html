100Double_t PFermiDistributions::Eval(Double_t x, Double_t, Double_t, Double_t) const {
101 // Returns connected fermi distribution
102
103 Double_t p = x/MeV2GeV;
104 if (version_flag == NUCLEAR_FERMI_D)
105 return FermiDisD(&p, NULL);
106 else if (version_flag == NUCLEAR_FERMI_3HE)
107 return FermiDis3He(&p, NULL);
108 else if (version_flag == NUCLEAR_FERMI_4HE)
109 return FermiDis4He(&p, NULL);
110 else if (version_flag == NUCLEAR_FERMI_7LI)
111 return FermiDis7Li(&p, NULL);
112 else if (version_flag == NUCLEAR_FERMI_12C)
113 return FermiDis12C(&p, NULL);
114 else if (version_flag == NUCLEAR_FERMI_40CA)
115 return FermiDis40Ca(&p, NULL);
116 return 0.;
117}

120//---------------------- Fermi Distribution of Deuteron -----------------------
121double PFermiDistributions::FermiDisD(double *x, double *) const {
122
123 const double pi = 3.1415927;
124 const double hbar = 0.197463569747999998; // unit: GeV*fm/c
125 double par0 = 0.;
126 double par1 = 0.;
127 for (int j=0; j<13; j++) {
128 par0 += DeutC[j]/(x[0]*x[0]/(hbar*hbar) + DeutM2[j]);
129 par1 += DeutD[j]/(x[0]*x[0]/(hbar*hbar) + DeutM2[j]);
130 }
131 par0 = TMath::Sqrt(2./pi)*par0;
132 par1 = TMath::Sqrt(2./pi)*par1;
133 Double_t mom = (par0*par0+par1*par1)*x[0]*x[0]/(hbar*hbar);
134
135 return mom;
136}

138//---------------------- Fermi Distribution of Helium-3 ---------------------
139double PFermiDistributions::FermiDis3He(double *x, double *) const {
140
141 const double pi = 3.1415927;
142 const double a = 7.09078;
143 const double b = 5.38753;
144 const double c = 9.90202;
145 const double d = 0.779408;
146
147 Double_t par0 = (4./TMath::Sqrt(pi))*TMath::Power(a,3./2.);
148 Double_t par1 = x[0]*x[0]*25./(1.E6);
149 Double_t mom = d*par0*par1*(TMath::Exp(-par1*a) + c*TMath::Exp(-TMath::Sqrt(par1)*b));
150
151 return mom;
152}

154//---------------------- Fermi Distribution of Helium-4 ---------------------
155double PFermiDistributions::FermiDis4He(double *x, double *) const {
156
157 const double hbar = 0.197463569747999998;
158 const double a = 0.7352;
159 const double b = 0.05511;
160
161 Double_t par0 = TMath::Exp(-x[0]*x[0]/(hbar*hbar*a));
162 Double_t mom = x[0]*x[0]/(hbar*hbar*b)*par0;
163
164 return mom;
165}

167//---------------------- Fermi Distribution of Lithium-7 ---------------------
168double PFermiDistributions::FermiDis7Li(double *x, double *) const {
169
170 const double a = 1.2e-4;
171 const double b = 6.87e-3;
172 const double c = 110.;
173
174 Double_t par0 = x[0]*x[0]*a;
175 Double_t par1 = (1. + b*x[0]);
176 Double_t par2 = x[0]/c;
177 Double_t mom = par0*par1*TMath::Exp(-par2*par2);
178
179 return mom;
180}

212//------ Calculate Values for Deuteron Wave Function ----------------
213void PFermiDistributions::SetDeuteronValues( ) {
214
215 const double alpha = 0.23162461; // unit: 1/fm
216 const double mZero = 1.; // unit: 1/fm
217 double cj[13]={ 0.88688076E0, // unit: 1/sqrt(fm)
218 -0.34717093E0,
219 -0.30502380E1,
220 0.56207766E2,
221 -0.74957334E3,
222 0.53365279E4,
223 -0.22706863E5,
224 0.60434469E5,
225 -0.10292058E6,
226 0.11223357E6,
227 -0.75925226E5,
228 0.29059715E5,
229 0.0};
230 double dj[13]={ 0.23135193E-1, // unit: 1/sqrt(fm)
231 -0.85604572E0,
232 0.56068193E1,
233 -0.69462922E2,
234 0.41631118E3,
235 -0.12546621E4,
236 0.12387830E4,
237 0.33739172E4,
238 -0.13041151E5,
239 0.19512524E5,
240 0.0,
241 0.0,
242 0.0};
243 double sum1 = 0.;
244 double sum2 = 0.;
245 double sum3 = 0.;
246 double rtmp;
247 double mj[13];
248 double mj2[13];
249 int n;
250 int n1;
251 int n2;
252 int temp;
253
254 for (int j=0; j<13; j++) {
255 mj[j] = alpha + j*mZero;
256 mj2[j] = mj[j]*mj[j];
257 }
258
259 cj[12] = 0.;
260 for (int i=0; i<12; i++)
261 cj[12] = cj[12] - cj[i];
262
263 for (int k=0; k<5; k++) {
264 rtmp = dj[k*2]/mj2[k*2] + dj[k*2+1]/mj2[k*2+1];
265 sum1 = sum1 + rtmp;
266 rtmp = dj[k*2] + dj[k*2+1];
267 sum2 = sum2 + rtmp;
268 rtmp = dj[k*2]*mj2[k*2] + dj[k*2+1]*mj2[k*2+1];
269 sum3 = sum3 + rtmp;
270 }
271
272 n = 12;
273 n1 = 11;
274 n2 = 10;
275
276 for (int l=0; l<3; l++) {
277 dj[n2] = -mj2[n1]*mj2[n]*sum1 + (mj2[n1]+mj2[n])*sum2 - sum3;
278 dj[n2] = dj[n2] * mj2[n2]/(mj2[n]-mj2[n2])/(mj2[n1]-mj2[n2]);
279
280 temp = n2;
281 n2 = n1;
282 n1 = n;
283 n = temp;
284 }
285
286 for (int j=0; j<13; j++) {
287 DeutC[j] = cj[j];
288 DeutD[j] = dj[j];
289 DeutM2[j] = mj2[j];
290 }
291}
