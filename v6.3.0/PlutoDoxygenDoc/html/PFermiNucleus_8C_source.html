51double PFermiNucleus::SampleFermi(double &px, double &py, double &pz) {
52 // dummy function
53
54 double p = 0.5; //150 MeV
55 //double p = 0.;
56 double theta = acos(1.-2.*PUtils::sampleFlat());
57 double phi = 2.*TMath::Pi()*PUtils::sampleFlat();
58 double sth=sin(theta);
59 px = p*cos(phi)*sth;
60 py = p*sin(phi)*sth;
61 pz = p*cos(theta);
62 return p;
63}

85Bool_t PFermiNucleus::SampleMass(void) {
86
87 Double_t massS, eS, eP, ptot, px, py, pz, t=-1., mtarget;
88 while (t<0.) {
89 ptot = SampleFermi(px,py,pz); // Fermi momentum
90 massS = spectator->M(); // mass of spectator nucleon
91 eS = sqrt(ptot*ptot + massS*massS); // spectator total energy in deuteron c.m.
92 mtarget = target->M();
93 t = pow(mtarget-massS,2) - 2.*mtarget*(eS-massS); // off-shell mass**2 of participant
94 }
95
96 eP = sqrt(ptot*ptot + t); // participant total energy
97
98 participant->SetPxPyPzE(-px,-py,-pz,eP); // initialize participant nucleon
99 spectator->SetPxPyPzE(px,py,pz,eS); // initialize spectator nucleon
100
101// participant->Print();
102// spectator->Print();
103
104 participant->Boost(target->BoostVector());
105 spectator->Boost(target->BoostVector());
106
107 *p2 = *beam;
108
109 //go into parent frame
110 participant->Boost(-parent->BoostVector());
111 p2->Boost(-parent->BoostVector());
112 spectator->Boost(-parent->BoostVector());
113
114 composite->Reconstruct(); //reset mass after p1 and p2 have been setted
115
116 //boost scatter back to lab
117 participant->Boost(parent->BoostVector());
118 p2->Boost(parent->BoostVector());
119
120 spectator->SetW(parent->W()); // copy parent weight to spectator
121
122 return kTRUE;
123}
