30double PResonanceDalitz::GetMatrixT(int &spin, int &parity, int&, const double& ecm, const double& M) {
31 float matr;
32 if (spin == 1) {
33 if (parity == 1) {
34 matr = TMath::Power((ecm2 - mn2),2)*(ecmPmn2 - M2);
35 return matr/2./mn4;
36 } else {
37 matr = TMath::Power((ecm2 - mn2),2)*(ecmMmn2 - M2);
38 return matr/2./mn4;
39 }
40 } else if (spin == 3) {
41 if (parity == 1) {
42 matr = (ecmMmn2 - M*M )*(3.*ecm4 + 6.*ecm3*mn + 4.*ecm2*mn2 +
43 2.*ecm*mn3 + mn4 - 2.*ecm*mn*M2 - 2.*mn2*M2 + M4);
44 return matr/12./ecm2/mn2;
45 } else {
46 matr = (ecmPmn2 - M*M )*(3.*ecm4 - 6.*ecm3*mn + 4.*ecm2*mn2 -
47 2.*ecm*mn3 + mn4 + 2.*ecm*mn*M2 - 2.*mn2*M2 + M4);
48 return matr/12./ecm2/mn2;
49 }
50 } else if (spin == 5) {
51 if (parity == 1) {
52 matr = (ecmMmn2 - M*M )*(ecmPmn2 - M*M )*(ecmPmn2 - M*M )*

54 2.*ecm*mn*M2 - 2.*mn2*M2 + M4);
55 return matr/480./ecm4/mn4;
56 } else {

59 2.*ecm*mn*M2 - 2.*mn2*M2 + M4);

106double PResonanceDalitz::dGdM(const int&, const double& m, const double& ecm) {
107
108 mn = 0.938273;
109 mn2 = mn*mn;
110 mn3 = mn*mn*mn;
111 mn4 = mn*mn*mn*mn;
112 ecm2 = TMath::Power(ecm,2);
113 ecm4 = TMath::Power(ecm,4);
114 ecm3 = TMath::Power(ecm,3);
115 ecmPmn2 = TMath::Power((ecm + mn),2);
116 ecmMmn2 = TMath::Power((ecm - mn),2);
117 M2 = m*m;
118 M4 = m*m*m*m;
119 double lam = getLambda(ecm*ecm,mn*mn,m*m);
120 double alfa = 0.007297;
121 double pi = 3.14159265;
122
123 double dgdm = alfa*alfa*g_Em*g_Em*sqrt(lam)*
124 (2.*GetMatrixT(spin,par,charge,ecm,m) +
125 GetMatrixL(spin,par,charge,ecm,m))/6./pi/ecm/ecm/ecm/m;
126
127 if (dgdm > 0) return dgdm;
128 return 0;
129}
