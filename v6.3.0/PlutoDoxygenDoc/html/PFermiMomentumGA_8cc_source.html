66Double_t PFermiMomentumGA::GetRandomFermiMomentum(Double_t &px, Double_t &py, Double_t &pz) {
67
68 //BUGBUG: What about the cross section
69 if (!fermi_model) {
70 //Read secondary fermi model
71 //This can only happen in Init, so after the freeze-out of the user
72 Int_t parent_id = makeStaticData()->GetDecayParentByKey(primary_key);
73
74 Int_t id1 = parent_id/1000;
75 Int_t id2 = parent_id%1000;
76 if (id1 > 600) {
77 fermi_model = makeDynamicData()->
78 GetParticleSecondaryModel(makeStaticData()->GetParticleName(id1), "fermi");
79 } else if (id2 > 600) {
80 fermi_model = makeDynamicData()->
81 GetParticleSecondaryModel(makeStaticData()->GetParticleName(id2), "fermi");
82 } else if (makeStaticData()->GetParticleBaryon(id1)
83 > makeStaticData()->GetParticleBaryon(id2)) {
84 fermi_model = makeDynamicData()->
85 GetParticleSecondaryModel(makeStaticData()->GetParticleName(id1), "fermi");
86 } else if (makeStaticData()->GetParticleBaryon(id2)
87 > makeStaticData()->GetParticleBaryon(id1)) {
88 fermi_model = makeDynamicData()->
89 GetParticleSecondaryModel(makeStaticData()->GetParticleName(id2), "fermi");
90 }
91
92 if (!fermi_model) {
93 Error("GetRandomFermiMomentum", "No fermi model found");
94 return 0.;
95 }
96 }
97
98 Double_t p = (fermi_model->GetRandom());
99 Double_t theta = acos(1.-2.*PUtils::sampleFlat());
100 Double_t phi = 2.*TMath::Pi()*PUtils::sampleFlat();
101 Double_t sth = sin(theta);
102
103 px = p*cos(phi)*sth;
104 py = p*sin(phi)*sth;
105 pz = p*cos(theta);
106
107 return p;
108}

110Bool_t PFermiMomentumGA::SampleMass(void) {
111
112 Double_t massS, eS, eP, ptot, px, py, pz, t=-1., mAi;
113
114 parent->Reconstruct();
115
116 while (t < 0.) {
117 ptot = GetRandomFermiMomentum(px,py,pz); // Fermi momentum
118 massS = spectator->M(); // mass of spectator fragment
119 eS = sqrt(ptot*ptot + massS*massS); // spectator total energy in nucleus c.m.
120 mAi = target->M();
121 t = pow(mAi-massS,2) - 2.*mAi*(eS-massS); // off-shell mass**2 of participant
122 }
123
124 eP = sqrt(ptot*ptot + t); // participant total energy
125
126 participant->SetPxPyPzE(-px,-py,-pz,eP); // initialize participant
127 spectator->SetPxPyPzE(px,py,pz,eS); // initialize spectator fragment
128
129 participant->Boost(target->BoostVector());
130 spectator->Boost(target->BoostVector());
131
132 *p2 = *beam;
133
134 //go into parent frame
135 participant->Boost(-parent->BoostVector());
136 p2->Boost(-parent->BoostVector());
137 spectator->Boost(-parent->BoostVector());
138
139 composite->Reconstruct();
140
141 //boost scatter back to lab
142 participant->Boost(parent->BoostVector());
143 p2->Boost(parent->BoostVector());
144
145 spectator->SetW(parent->W()); // copy parent weight to spectator fragment
146
147 return kTRUE;
148}
