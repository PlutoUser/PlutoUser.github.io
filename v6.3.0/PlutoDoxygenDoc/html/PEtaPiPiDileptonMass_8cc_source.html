27PEtaPiPiDileptonMass::PEtaPiPiDileptonMass(const Char_t *id, const Char_t *de, Int_t key) :

47Bool_t PEtaPiPiDileptonMass::Init(void) {

91Double_t PEtaPiPiDileptonMass::GetMassWeight(Double_t mass) const {
92
93 //Mass-dependent part of the eqn. 64
94
95 if (mass < mass_ee) return 0.;
96
97 Double_t weight = 1./(8*mass*mass);
98
99 if (vmd_formfactor_model) {
100 weight *= vmd_formfactor_model->GetWeight(mass);
101 }
102
103 return weight;
104
105}
