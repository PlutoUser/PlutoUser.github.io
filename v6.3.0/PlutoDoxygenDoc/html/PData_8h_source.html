32 static double LMass(const int &id) {
33 // lower mass limit, by particle pid
34
35 Double_t *mass;
36
37 if (makeDataBase()->GetParamDouble ((char*)"pid", id, (char*)"lmass", &mass)) {
38 return *mass;
39 }
40
41 if ( makeStaticData()->GetParticleTotalWidth(id) < 1.e-3 ) {
42 // (quasi)stable
43 return 0.999*makeStaticData()->GetParticleMass(id);
44 }
45// else if ( id==41 ) return 2.*PMass[8]; // rho0
46 else if (id == 41) // rho0
47 return 2.*makeStaticData()->GetParticleMass(2);
48 else if (id == 42 || id == 43) // rho+,rho-
49 return makeStaticData()->GetParticleMass(8)+
50 makeStaticData()->GetParticleMass(7);
51 else if (id == 52) // omega
52 return 2.*makeStaticData()->GetParticleMass(2);
53 else if (id == 55) // phi
54 return 3.*makeStaticData()->GetParticleMass(2);
55 else if (makeStaticData()->GetParticleBaryon(id)) // resonance
56 return makeStaticData()->GetParticleMass(14);
57 else if (makeStaticData()->GetParticleTotalWidth(id) > 0.01)
58 return TMath::Max(0., 0.999*(makeStaticData()->GetParticleMass(id)-
59 5.*makeStaticData()->GetParticleTotalWidth(id)));
60 else
61 return TMath::Max(0., 0.999*(makeStaticData()->GetParticleMass(id)-
62 15.*makeStaticData()->GetParticleTotalWidth(id)));
63 }
